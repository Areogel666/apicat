## 关键约束

- **变量 JSON 注入**（设计文档 §10.3 CRITICAL）：`body_type = raw_json` 时，变量替换必须对值进行 JSON 字符串转义（`serde_json::to_string` + `trim_matches('"')`）
- **Cookie 优先级**：项目级 Cookie > 全局 Cookie（同域同名同 path 时项目级覆盖全局级）
- **环境激活互斥**：每个项目同时只有一个激活环境（`is_active = 1`），切换时需先 deactivate 旧环境
- **Windows MSVC 编译**：每次 Rust 修改后用 VsDevCmd cargo check 命令验证

/// 获取指定域名匹配的 Cookie（发请求时调用）
/// 返回优先级：项目级 > 全局，同名同 path 的项目级 Cookie 覆盖全局
#[tauri::command]
pub async fn get_cookies_for_domain(
    db: State<'_, AppDb>,
    domain: String,
    project_id: Option<i64>,
) -> CmdResult<Vec<Cookie>> {
    // 先查全局 + 项目级所有匹配域名的 enabled Cookie
    let all = sqlx::query_as::<_, Cookie>(
        r#"SELECT id, scope_type, project_id, domain, name, value, path, expires_at, http_only, secure, enabled
           FROM cookies
           WHERE domain = ? AND enabled = 1
             AND (scope_type = 'global' OR (scope_type = 'project' AND project_id = ?))
           ORDER BY scope_type DESC"#,  -- project 排在 global 前面
    )
    .bind(&domain)
    .bind(project_id)
    .fetch_all(&db.0)
    .await?;

    // 去重：同 name + path，项目级优先（已排序，first wins）
    let mut seen = std::collections::HashSet::new();
    let mut result = Vec::new();
    for c in all {
        let key = format!("{}:{}", c.name, c.path);
        if seen.insert(key) {
            result.push(c);
        }
    }
    Ok(result)
}

use std::collections::HashMap;

/// 替换字符串中的 {{variable}} 占位符
/// 当 is_json_body 为 true 时，对值进行 JSON 字符串转义（设计文档 §10.3 CRITICAL）
pub fn replace_variables(
    input: &str,
    variables: &HashMap<String, String>,
    is_json_body: bool,
) -> String {
    if variables.is_empty() || !input.contains("{{") {
        return input.to_string();
    }

    let mut result = input.to_string();
    for (key, value) in variables {
        let placeholder = format!("{{{{{}}}}}", key); // 生成 {{key}}
        let replacement = if is_json_body {
            // JSON 转义：处理双引号、反斜杠、换行等特殊字符（§10.3）
            serde_json::to_string(value)
                .unwrap_or_else(|_| value.clone())
                .trim_matches('"')
                .to_string()
        } else {
            value.clone()
        };
        result = result.replace(&placeholder, &replacement);
    }
    result
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_basic_replacement() {
        let mut vars = HashMap::new();
        vars.insert("base_url".to_string(), "https://api.example.com".to_string());
        let result = replace_variables("{{base_url}}/users", &vars, false);
        assert_eq!(result, "https://api.example.com/users");
    }

    #[test]
    fn test_json_escape() {
        let mut vars = HashMap::new();
        vars.insert("name".to_string(), r#"foo"bar"#.to_string());
        // 在 JSON body 中，双引号应被转义
        let result = replace_variables(r#"{"name": "{{name}}"}"#, &vars, true);
        assert_eq!(result, r#"{"name": "foo\"bar"}"#);
    }

    #[test]
    fn test_no_variables() {
        let vars = HashMap::new();
        let result = replace_variables("hello world", &vars, false);

    }

    #[test]

        vars.insert("port".to_string(), "8080".to_string());
        let result = replace_variables("http://{{host}}:{{port}}/api", &vars, false);
        assert_eq!(result, "http://localhost:8080/api");
    }
}

在 `send_request.rs` 中：
1. 接收 `env_id: Option<i64>` 和 `project_id: Option<i64>` 参数
2. 若有 env_id，查询该环境所有 enabled 变量，构建 HashMap
3. 对 params.url / params.headers / params.body 执行变量替换
4. 查询匹配域名的 Cookie，注入到请求 headers

关键代码变更：

```rust
// send_request.rs — 新增 imports
use crate::http::variable::replace_variables;
use std::collections::HashMap;

#[tauri::command]
pub async fn send_request(
    db: State<'_, AppDb>,
    http: State<'_, HttpClient>,
    request_id: i64,
    params: SendRequestParams,
    env_id: Option<i64>,        // 新增：当前激活环境 ID
    project_id: Option<i64>,    // 新增：当前项目 ID（用于查 Cookie）
) -> CmdResult<HttpResponse> {
    // ── 1. 加载环境变量 ──────────────────────────────────────
    let variables: HashMap<String, String> = if let Some(eid) = env_id {
        let vars = sqlx::query_as::<_, crate::types::EnvVariable>(
            "SELECT id, env_id, key, value, description, enabled FROM env_variables WHERE env_id = ? AND enabled = 1"
        )
        .bind(eid)
        .fetch_all(&db.0)
        .await?;

        // 如果有 base_url，也加入变量表
        let env = sqlx::query_as::<_, crate::types::Environment>(
            "SELECT id, project_id, name, base_url, is_active, created_at FROM environments WHERE id = ?"
        )
        .bind(eid)
        .fetch_one(&db.0)
        .await?;

        let mut map: HashMap<String, String> = vars.into_iter().map(|v| (v.key, v.value)).collect();
        if let Some(base) = env.base_url {
            map.entry("base_url".to_string()).or_insert(base);
        }
        map
    } else {
        HashMap::new()
    };

    // ── 2. 变量替换 ─────────────────────────────────────────
    let is_json = params.body_type == "raw_json";
    let mut resolved_params = params.clone();  // 注意：需要给 SendRequestParams 加 Clone
    resolved_params.url = replace_variables(&params.url, &variables, false);
    resolved_params.body = replace_variables(&params.body, &variables, is_json);
    // Headers 值也替换
    for h in resolved_params.headers.iter_mut() {
        h.value = replace_variables(&h.value, &variables, false);
    }

    // ── 3. 加载 Cookie 并注入 ────────────────────────────────
    if let Ok(parsed_url) = reqwest::Url::parse(&resolved_params.url) {
        if let Some(domain) = parsed_url.host_str() {
            let cookies = sqlx::query_as::<_, crate::types::Cookie>(
                r#"SELECT id, scope_type, project_id, domain, name, value, path, expires_at, http_only, secure, enabled
                   FROM cookies WHERE domain = ? AND enabled = 1
                   AND (scope_type = 'global' OR (scope_type = 'project' AND project_id = ?))
                   ORDER BY scope_type DESC"#,
            )
            .bind(domain)
            .bind(project_id)
            .fetch_all(&db.0)
            .await
            .unwrap_or_default();

            // 去重（项目级优先）后拼接 Cookie header
            let mut seen = std::collections::HashSet::new();
            let cookie_pairs: Vec<String> = cookies
                .into_iter()
                .filter(|c| seen.insert(format!("{}:{}", c.name, c.path)))
                .map(|c| format!("{}={}", c.name, c.value))
                .collect();

            if !cookie_pairs.is_empty() {
                resolved_params.headers.push(crate::http::client::ParamItem {
                    key: "Cookie".to_string(),
                    value: cookie_pairs.join("; "),
                    enabled: true,
                });
            }
        }
    }

    // ── 4. 发送请求（原逻辑不变）────────────────────────────
    let mut resp = send(&http.0, &resolved_params).await.map_err(crate::error::AppError::Custom)?;
    // ... 后续 snapshot / history 写入不变 ...
}
```
